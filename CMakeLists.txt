cmake_minimum_required(VERSION 3.20)
project(vortex VERSION 0.1.0 LANGUAGES CXX)
option(VORTEX_PYTHON "Build Python extension" OFF)
option(VORTEX_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(vortex src/distance.cpp src/storage.cpp src/index.cpp)
target_include_directories(vortex PUBLIC include)
target_link_libraries(vortex PUBLIC Threads::Threads)
if(MINGW)
  target_link_options(vortex PUBLIC -static -static-libgcc -static-libstdc++)
endif()
set_target_properties(vortex PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(vortex PRIVATE /W4 /permissive-)
else()
  target_compile_options(vortex PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
if(VORTEX_SANITIZERS AND WIN32)
  message(FATAL_ERROR "This sanitizer configuration requires Linux GCC/Clang; use the sanitizer CI job.")
endif()
if(VORTEX_SANITIZERS)
  target_compile_options(vortex PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(vortex PUBLIC -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(vortex_tests tests/core_tests.cpp)
  target_link_libraries(vortex_tests PRIVATE vortex)
  add_test(NAME core COMMAND vortex_tests)
endif()
add_executable(vortex_bench benchmarks/microbench.cpp)
target_link_libraries(vortex_bench PRIVATE vortex)
if(VORTEX_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
    OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_vortex python/bindings.cpp)
  target_link_libraries(_vortex PRIVATE vortex)
  set_target_properties(_vortex PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/python")
  foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(_vortex PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${cfg} "${CMAKE_BINARY_DIR}/python"
      RUNTIME_OUTPUT_DIRECTORY_${cfg} "${CMAKE_BINARY_DIR}/python")
  endforeach()
  install(TARGETS _vortex LIBRARY DESTINATION . RUNTIME DESTINATION .)
endif()
